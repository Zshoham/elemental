cmake_minimum_required(VERSION 3.22)
project(portable_check LANGUAGES C)

add_subdirectory("${ELEMENTAL_SOURCE_DIR}" elemental EXCLUDE_FROM_ALL)
add_library(checked STATIC caller.c)
target_link_libraries(checked PRIVATE elemental::fmt)
set_property(TARGET checked PROPERTY C_COMPILER_LAUNCHER "${CMAKE_COMMAND};-E;env")
target_compile_definitions(checked PRIVATE CHECK_THIS_CALL=1)
set_source_files_properties(caller.c PROPERTIES COMPILE_DEFINITIONS "LOCAL_FORMAT=\"value %f\"")
if(CHECK_LIBC)
  elemental_check_formats(checked LIBC FUNCTIONS report:2)
else()
  elemental_check_formats(checked FUNCTIONS report:2)
endif()
